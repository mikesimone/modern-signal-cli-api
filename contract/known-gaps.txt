# Calls our test-suite makes that the real signal-cli would not honour, each with the reason it is accepted for now.
# One `method` (signal-cli has no such method) or `method:param` (param is silently ignored/rejected) per line.
# check.py fails on anything NOT listed here, and on entries here that no longer occur - so this can only shrink.

# --- routes with no signal-cli equivalent: they always return an error ---
deleteAttachment
listAttachments
getConfiguration
setConfiguration
getAccountSettings
setAccountSettings

# --- joinGroup needs a group invite `uri`; the route has only a group id, so it cannot work ---
joinGroup:group-id

# --- sendReceipt has no group parameter in signal-cli (receipts go to a recipient); a `group-id` in the body is ignored.
#     (Legacy reaction/receipt/remote-delete body names are translated to signal-cli's - see helpers.rs.) ---
sendReceipt:group-id

# --- /v2/send is a deliberate pure passthrough: these fields come from the tests, which check that unknown or
#     bbernhard-style fields are forwarded verbatim. They are not what our code generates. ---
send:number
send:base64_attachments
send:quote_timestamp
send:quote_author
send:quote_message
send:somethingSignalCliMightAddTomorrow

# --- signal-cli's startLink takes no params (the device name is given later, to finishLink): the qrcodelink
#     route's device_name is accepted for compatibility but has no effect ---
startLink:deviceName
